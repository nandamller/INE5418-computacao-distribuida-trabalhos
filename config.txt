# Configuração do Encurtador de URLs Distribuído
# Formato: chave=valor   (linhas começando com # são ignoradas)
# Variáveis de ambiente sobrescrevem estes valores quando definidas.

# --- Servidor REST ---
server_host=0.0.0.0
server_port=5000

# --- Interceptador (proxy) ---
proxy_host=0.0.0.0
proxy_port=8080

# URL que o interceptador usa para alcançar o servidor REST
rest_server_url=http://server:5000/urls

# --- Cliente: endereço do interceptador ---
# Em docker-compose, "proxy" é o nome do serviço.
# Para rodar fora do docker, sobrescreva via env PROXY_HOST=localhost
client_proxy_host=proxy
client_proxy_port=8080

# --- Cache LRU ---
cache_capacity=5
